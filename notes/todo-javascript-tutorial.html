搭出输入框和列表

新建一个练习文件夹，里面放 index.html 和 todo.js。先把下面的代码写入 index.html。

<!doctype html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <title>我的待办清单</title>
</head>
<body>
  <h1>我的待办清单</h1>
  <form id="task-form">
    <label for="task-text">新任务</label>
    <input id="task-text" required>
    <button type="submit">添加</button>
  </form>
  <ul id="task-list"></ul>
  <script src="./todo.js"></script>
</body>
</html>

表单接收输入，ul 展示任务。按回车和点击「添加」都会触发表单的 submit 事件。

用数组记录任务，再更新列表

每条任务用三个字段表示。id 是唯一编号，text 是任务文字，done 记录是否完成。即使两条任务都叫「整理笔记」，删除时也可以靠编号选中正确的一条。

把下面的代码写入 todo.js。

const form = document.querySelector('#task-form');
const input = document.querySelector('#task-text');
const list = document.querySelector('#task-list');

let tasks = [];

function updateTasks(nextTasks) {
  tasks = nextTasks;
  render();
}

function render() {
  const rows = tasks.map(task => {
    const row = document.createElement('li');
    const label = document.createElement('label');
    const checkbox = document.createElement('input');
    const text = document.createElement('span');
    const remove = document.createElement('button');

    checkbox.type = 'checkbox';
    checkbox.checked = task.done;
    text.textContent = task.text;

    checkbox.addEventListener('change', () => {
      updateTasks(tasks.map(item => item.id === task.id
        ? { ...item, done: checkbox.checked }
        : item));
    });

    remove.type = 'button';
    remove.textContent = '删除';
    remove.setAttribute('aria-label', '删除任务 ' + task.text);
    remove.addEventListener('click', () => {
      updateTasks(tasks.filter(item => item.id !== task.id));
    });

    label.append(checkbox, text);
    row.append(label, remove);
    return row;
  });

  list.replaceChildren(...rows);
}

form.addEventListener('submit', event => {
  event.preventDefault();
  const text = input.value.trim();
  if (!text) return;

  updateTasks([...tasks, { id: crypto.randomUUID(), text, done: false }]);
  input.value = '';
  input.focus();
});

render();

完整项目的分类、搜索和撤销逻辑，可以对照 待办清单项目案例。练习里的三种操作都调用 updateTasks()。添加时把新任务放进数组，勾选时更新对应任务的 done，删除时过滤掉对应的 id。render() 根据数组重建列表。任务文字用 textContent 写入，所以输入 <b>整理笔记</b> 会原样显示为文字。

打开页面做一次检查

用 VS Code 的本地预览打开页面。电脑上有 Python 的话，也可以在练习文件夹的终端执行下面的命令，再访问 http://127.0.0.1:8000/。

python -m http.server 8000 --bind 127.0.0.1
  1. 添加「阅读文章」和「整理笔记」。
  2. 勾选第一条，再删除第二条。应只留下已勾选的「阅读文章」。
  3. 输入几个空格并提交，列表应保持原样。

刷新后清单会变空，因为每次加载脚本都会重新执行 let tasks = []。接着用 localStorage 保存任务并在刷新后恢复，把这个小工具变成实际能用的清单。
